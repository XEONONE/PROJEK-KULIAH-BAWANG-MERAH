# ==============================================================================
# PROYEK SKRIPSI: KLASIFIKASI MUTU BAWANG MERAH (SNI 3159:2013)
# METODE: CNN MURNI vs (GLCM + SVM)

import os
import cv2
import numpy as np
import seaborn as sns
import matplotlib.pyplot as plt
from google.colab import drive

from skimage.feature import graycomatrix, graycoprops
from sklearn.metrics import accuracy_score, confusion_matrix, classification_report
from sklearn.svm import SVC
from sklearn.utils.class_weight import compute_class_weight

from tensorflow.keras.preprocessing.image import ImageDataGenerator
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Conv2D, MaxPooling2D, Flatten, Dense, Dropout, Input

# 1. MOUNT GOOGLE DRIVE
drive.mount('/content/drive')

# SESUAIKAN JALUR FOLDER DRIVE LO DI SINI
path_dataset = '/content/drive/MyDrive/MACHINE LEARNING VISUAL TEXTURE'
path_train = os.path.join(path_dataset, 'train')
path_valid = os.path.join(path_dataset, 'valid')
path_test = os.path.join(path_dataset, 'test')

# 2. DATA GENERATOR (Persiapan Gambar untuk CNN)
datagen = ImageDataGenerator(rescale=1./255)

print("\n[INFO] Menyiapkan Data Training...")
train_generator = datagen.flow_from_directory(
path_train, target_size=(224, 224), batch_size=32, class_mode='categorical')

valid_generator = datagen.flow_from_directory(
path_valid, target_size=(224, 224), batch_size=32, class_mode='categorical')

test_generator = datagen.flow_from_directory(
path_test, target_size=(224, 224), batch_size=32, class_mode='categorical', shuffle=False)

class_names = list(train_generator.class_indices.keys())
num_classes = len(class_names)

# ---> SOLUSI DATA JOMPLANG: MENGHITUNG CLASS WEIGHTS <---
class_weights_array = compute_class_weight(
class_weight='balanced',
classes=np.unique(train_generator.classes),
y=train_generator.classes
)
class_weights_dict = dict(enumerate(class_weights_array))
print("\n[INFO] Bobot Penyeimbang Data Jomplang (Class Weights):", class_weights_dict)

# 3. MEMBANGUN DAN MELATIH MODEL CNN
print("\n[INFO] Membangun Arsitektur CNN...")
model_cnn = Sequential([
Input(shape=(224, 224, 3)),
Conv2D(32, (3,3), activation='relu'),
MaxPooling2D(2, 2),
Conv2D(64, (3,3), activation='relu'),
MaxPooling2D(2,2),
Conv2D(128, (3,3), activation='relu'),
MaxPooling2D(2,2),
Flatten(),
Dense(128, activation='relu'),
Dropout(0.5),
Dense(num_classes, activation='softmax')
])

model_cnn.compile(loss='categorical_crossentropy', optimizer='adam', metrics=['accuracy'])

print("\n=== MEMULAI TRAINING MODEL CNN (15 EPOCHS) ===")
history = model_cnn.fit(
train_generator,
epochs=15,
validation_data=valid_generator,
class_weight=class_weights_dict,
verbose=1
)

# Evaluasi CNN
test_generator.reset()
pred_cnn_prob = model_cnn.predict(test_generator)
pred_cnn_classes = np.argmax(pred_cnn_prob, axis=1)
y_true_cnn = test_generator.classes
acc_cnn = accuracy_score(y_true_cnn, pred_cnn_classes) * 100

# 4. EKSTRAKSI FITUR GLCM & PELATIHAN SVM
print("\n=== MEMULAI EKSTRAKSI FITUR GLCM UNTUK SVM ===")

def ekstraksi_glcm(image_path):
img = cv2.imread(image_path)
if img is None: return None
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
gray = cv2.resize(gray, (224, 224))
angles = [0, np.pi/4, np.pi/2, 3*np.pi/4]
glcm = graycomatrix(gray, distances=[1], angles=angles, levels=256, symmetric=True, normed=True)
contrast = np.mean(graycoprops(glcm, 'contrast'))
correlation = np.mean(graycoprops(glcm, 'correlation'))
energy = np.mean(graycoprops(glcm, 'energy'))
homogeneity = np.mean(graycoprops(glcm, 'homogeneity'))
return [contrast, correlation, energy, homogeneity]

def load_data_glcm(path):
X, y = [], []
for label in sorted(os.listdir(path)):
folder = os.path.join(path, label)
if os.path.isdir(folder):
for f in os.listdir(folder):
fitur = ekstraksi_glcm(os.path.join(folder, f))
if fitur:
X.append(fitur)
y.append(label)
return np.array(X), np.array(y)

X_train_glcm, y_train_glcm = load_data_glcm(path_train)
X_test_glcm, y_test_glcm = load_data_glcm(path_test)

print("\n[INFO] Melatih Model SVM dengan data GLCM...")
model_svm = SVC(kernel='linear', class_weight='balanced', random_state=42)
model_svm.fit(X_train_glcm, y_train_glcm)

pred_svm = model_svm.predict(X_test_glcm)
acc_glcm = accuracy_score(y_test_glcm, pred_svm) * 100

# 5. HASIL AKHIR & CONFUSION MATRIX
print("\n" + "="*50)
print(f">>> AKURASI UJIAN AKHIR CNN MURNI : {acc_cnn:.2f}% <<<")
print(f">>> AKURASI UJIAN AKHIR GLCM + SVM: {acc_glcm:.2f}% <<<")
print("="*50)

cm_cnn = confusion_matrix(y_true_cnn, pred_cnn_classes)
cm_glcm = confusion_matrix(y_test_glcm, pred_svm)

plt.figure(figsize=(12, 5))
plt.subplot(1, 2, 1)
sns.heatmap(cm_cnn, annot=True, fmt='d', cmap='Blues', xticklabels=class_names, yticklabels=class_names)
plt.title(f'Confusion Matrix - CNN\n(Akurasi: {acc_cnn:.2f}%)')
plt.xlabel('Tebakan Komputer')
plt.ylabel('Kenyataan Asli')

plt.subplot(1, 2, 2)
sns.heatmap(cm_glcm, annot=True, fmt='d', cmap='Oranges', xticklabels=np.unique(y_test_glcm), yticklabels=np.unique(y_test_glcm))
plt.title(f'Confusion Matrix - GLCM + SVM\n(Akurasi: {acc_glcm:.2f}%)')
plt.xlabel('Tebakan Komputer')
plt.ylabel('Kenyataan Asli')
plt.tight_layout()
plt.show()

print("\n--- REPORT CNN ---")
print(classification_report(y_true_cnn, pred_cnn_classes, target_names=class_names, zero_division=0))
print("\n--- REPORT GLCM + SVM ---")
print(classification_report(y_test_glcm, pred_svm, zero_division=0))
