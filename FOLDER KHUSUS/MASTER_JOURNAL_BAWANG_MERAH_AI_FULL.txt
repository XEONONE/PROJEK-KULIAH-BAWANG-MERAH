Bawang merah (Allium ascalonicum L.) merupakan salah satu komoditas hortikultura terpenting
di Indonesia dengan produksi nasional lebih dari 1,8 juta ton per tahun (BPS, 2023).

MASALAH UTAMA YANG DITEMUKAN DI LAPANGAN:
------------------------------------------
Indonesia masih melakukan sortasi mutu bawang merah secara MANUAL oleh tenaga manusia.
Metode ini mengalami tiga kelemahan fatal:

  1. SUBJEKTIF   : Hasil sortasi berbeda tiap operator / tiap hari
  2. LAMBAT      : Tidak bisa memproses volume besar saat panen raya
  3. TIDAK BAKU  : Tidak ada jaminan kesesuaian dengan SNI 3159:2013

STANDAR YANG HARUS DIPENUHI — SNI 3159:2013 (Badan Standardisasi Nasional):
  - Mutu I       : Umbi besar, kering, bersih, kulit rapat, tidak bertunas
  - Mutu II      : Umbi sedang, sedikit cacat minor yang masih bisa diterima pasar
  - Tidak Layak  : Busuk, bertunas, berjamur, rusak parah, tidak memenuhi SNI

SOLUSI YANG DITAWARKAN — Sistem Computer Vision berbasis AI:
  > CNN (Convolutional Neural Network)   : Deep Learning untuk klasifikasi visual
  > GLCM + SVM                           : Machine Learning tekstur + klasifier
  > Keduanya diintegrasikan ke aplikasi desktop real-time berbasis Python-Tkinter

     |

  +------------------------------------------+
  | Conv2D(32, (3,3), activation='relu')     | -> Feature map 32 channel
  | MaxPooling2D(2, 2)                       | -> Downsample 1/2
  +------------------------------------------+
     |
     v
  +------------------------------------------+
  | Conv2D(64, (3,3), activation='relu')     | -> Feature map 64 channel
  | MaxPooling2D(2, 2)                       | -> Downsample 1/2
  +------------------------------------------+
     |
     v
  +------------------------------------------+
  | Conv2D(128, (3,3), activation='relu')    | -> Feature map 128 channel
  | MaxPooling2D(2, 2)                       | -> Downsample 1/2
  +------------------------------------------+

     v
  +------------------------------------------+
  | Flatten()                                | -> 1D vector
  | Dense(128, activation='relu')            | -> Fully connected
  | Dropout(0.5)                             | -> Regularisasi 50%
  | Dense(3, activation='softmax')           | -> Output 3 kelas
  +------------------------------------------+
     |

# ================================================================================

# CELL 1: Import Library
from google.colab import drive
import os, cv2, numpy as np, seaborn as sns, matplotlib.pyplot as plt
from skimage.feature import graycomatrix, graycoprops
from sklearn.metrics import accuracy_score, confusion_matrix, classification_report
from sklearn.svm import SVC
from sklearn.utils.class_weight import compute_class_weight
from tensorflow.keras.preprocessing.image import ImageDataGenerator
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Conv2D, MaxPooling2D, Flatten, Dense, Dropout, Input
import joblib

# CELL 2: Mount Google Drive dan Persiapan Dataset
drive.mount('/content/drive', force_remount=True)
path_dataset = '/content/drive/MyDrive/MACHINE LEARNING VISUAL TEXTURE REVISI'
path_train   = os.path.join(path_dataset, 'train')
path_valid   = os.path.join(path_dataset, 'valid')
path_test    = os.path.join(path_dataset, 'test')

datagen         = ImageDataGenerator(rescale=1./255)
train_generator = datagen.flow_from_directory(path_train, target_size=(224, 224),
                                               batch_size=32, class_mode='categorical')
valid_generator = datagen.flow_from_directory(path_valid, target_size=(224, 224),
                                               batch_size=32, class_mode='categorical')
test_generator  = datagen.flow_from_directory(path_test, target_size=(224, 224),
                                               batch_size=32, class_mode='categorical',
                                               shuffle=False)
class_names = list(train_generator.class_indices.keys())
num_classes = len(class_names)
print(f"[INFO] Kelas: {class_names}")

# CELL 3: Hitung Class Weights
class_weights_array = compute_class_weight(class_weight='balanced',
                                            classes=np.unique(train_generator.classes),
                                            y=train_generator.classes)
class_weights_dict = dict(enumerate(class_weights_array))
print("[INFO] Class Weights:", class_weights_dict)

# CELL 4: Bangun Arsitektur CNN Sequential
model_cnn = Sequential([
    Input(shape=(224, 224, 3)),
    Conv2D(32, (3,3), activation='relu'),
    MaxPooling2D(2, 2),
    Conv2D(64, (3,3), activation='relu'),
    MaxPooling2D(2, 2),
    Conv2D(128, (3,3), activation='relu'),
    MaxPooling2D(2, 2),
    Flatten(),
    Dense(128, activation='relu'),
    Dropout(0.5),
    Dense(num_classes, activation='softmax')
])
model_cnn.compile(loss='categorical_crossentropy', optimizer='adam', metrics=['accuracy'])
model_cnn.summary()

# CELL 5: Training CNN selama 15 Epoch
history = model_cnn.fit(
    train_generator,
    epochs=15,
    validation_data=valid_generator,
    class_weight=class_weights_dict,
    verbose=1
)

# CELL 6: Evaluasi CNN pada Data Uji
test_generator.reset()
pred_cnn_prob    = model_cnn.predict(test_generator)
pred_cnn_classes = np.argmax(pred_cnn_prob, axis=1)
y_true_cnn       = test_generator.classes
acc_cnn          = accuracy_score(y_true_cnn, pred_cnn_classes) * 100
print(f"[HASIL] Akurasi CNN: {acc_cnn:.2f}%")
print(classification_report(y_true_cnn, pred_cnn_classes, target_names=class_names))

# CELL 7: Fungsi Ekstraksi Fitur GLCM (dengan CLAHE)
def ekstraksi_glcm(image_path):
    img   = cv2.imread(image_path)
    if img is None: return None
    gray  = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    gray  = cv2.resize(gray, (224, 224))
    clahe = cv2.createCLAHE(clipLimit=2.0, tileGridSize=(8, 8))
    gray  = clahe.apply(gray)
    glcm  = graycomatrix(gray, distances=[1], angles=[0], levels=256,
                         symmetric=True, normed=True)
    return [
        float(graycoprops(glcm, 'contrast')[0, 0]),
        float(graycoprops(glcm, 'correlation')[0, 0]),
        float(graycoprops(glcm, 'energy')[0, 0]),
        float(graycoprops(glcm, 'homogeneity')[0, 0])
    ]

def load_data_glcm(path):
    X, y = [], []
    for label in sorted(os.listdir(path)):
        folder = os.path.join(path, label)
        if os.path.isdir(folder):
            for fname in os.listdir(folder):
                fitur = ekstraksi_glcm(os.path.join(folder, fname))
                if fitur:
                    X.append(fitur)
                    y.append(label)
    return np.array(X), np.array(y)

# CELL 8: Training GLCM+SVM
X_train_glcm, y_train_glcm = load_data_glcm(path_train)
X_test_glcm,  y_test_glcm  = load_data_glcm(path_test)
model_svm = SVC(kernel='linear', class_weight='balanced', random_state=42)
model_svm.fit(X_train_glcm, y_train_glcm)
pred_svm  = model_svm.predict(X_test_glcm)
acc_glcm  = accuracy_score(y_test_glcm, pred_svm) * 100
print(f"[HASIL] Akurasi GLCM+SVM: {acc_glcm:.2f}%")
print(classification_report(y_test_glcm, pred_svm))

# CELL 9: Plot Confusion Matrix & Training History
cm_cnn  = confusion_matrix(y_true_cnn, pred_cnn_classes)
cm_glcm = confusion_matrix(y_test_glcm, pred_svm)
fig, axes = plt.subplots(1, 2, figsize=(14, 6))
sns.heatmap(cm_cnn, annot=True, fmt='d', cmap='Blues', ax=axes[0],
            xticklabels=class_names, yticklabels=class_names)
axes[0].set_title(f'Confusion Matrix CNN\nAkurasi: {acc_cnn:.2f}%')
sns.heatmap(cm_glcm, annot=True, fmt='d', cmap='Oranges', ax=axes[1],
            xticklabels=np.unique(y_test_glcm), yticklabels=np.unique(y_test_glcm))
axes[1].set_title(f'Confusion Matrix GLCM+SVM\nAkurasi: {acc_glcm:.2f}%')
plt.tight_layout()
plt.savefig('/content/drive/MyDrive/confusion_matrix_comparison.png', dpi=150)
plt.show()

# CELL 10: Simpan Model
model_cnn.save('/content/drive/MyDrive/Model_CNN_Bawang_Final.h5')
print("[INFO] Model CNN (.h5) berhasil disimpan!")
joblib.dump(model_svm, '/content/drive/MyDrive/Model_SVM_GLCM.pkl')
print("[INFO] Model SVM (.pkl) berhasil disimpan!")
import json
with open('/content/drive/MyDrive/class_names.json', 'w') as f:
    json.dump(class_names, f)
print(f"[INFO] class_names.json disimpan: {class_names}")

import tkinter as tk

MODEL_CNN_PATH = "Model_CNN_Bawang_Final.h5"
MODEL_SVM_PATH = "Model_SVM_GLCM.pkl"
CLASS_JSON     = "class_names.json"
IMG_SIZE       = (224, 224)

model_cnn  = tf.keras.models.load_model(MODEL_CNN_PATH)
model_svm  = joblib.load(MODEL_SVM_PATH)
with open(CLASS_JSON) as f:
    class_names = json.load(f)

def predict_cnn(path):
    img = cv2.resize(cv2.cvtColor(cv2.imread(path), cv2.COLOR_BGR2RGB), IMG_SIZE)
    pred = model_cnn.predict(np.expand_dims(img/255.0, 0), verbose=0)
    idx  = np.argmax(pred[0])
    return class_names[idx], float(pred[0][idx]) * 100

def predict_glcm_svm(path):
    gray  = cv2.resize(cv2.cvtColor(cv2.imread(path), cv2.COLOR_BGR2GRAY), IMG_SIZE)
    gray  = cv2.createCLAHE(clipLimit=2.0, tileGridSize=(8,8)).apply(gray)
    glcm  = graycomatrix(gray, distances=[1], angles=[0], levels=256,
                         symmetric=True, normed=True)
    fitur = [[float(graycoprops(glcm, p)[0,0]) for p in
              ['contrast','correlation','energy','homogeneity']]]
    return model_svm.predict(fitur)[0]

                                  fg="#aaaaaa", font=("Arial",11), width=40, height=18)
        self.img_label.pack(padx=10, pady=10)

        right = tk.Frame(main, bg="#1a1a2e"); right.pack(side='right', fill='both', expand=True, padx=5)
        ls = {"bg":"#1a1a2e","fg":"#e2b96f","font":("Arial",12,"bold")}
        vs = {"bg":"#1a1a2e","fg":"#ffffff","font":("Arial",13),"wraplength":320}

        for lbl, attr in [("Hasil CNN:", "cnn_result"), ("Keyakinan CNN:", "cnn_conf"),
                           ("Hasil GLCM+SVM:", "svm_result"), ("Waktu Inferensi:", "time_label")]:
            tk.Label(right, text=lbl, **ls).pack(anchor='w', pady=(10,0))
            setattr(self, attr, tk.Label(right, text="---", **vs))
            getattr(self, attr).pack(anchor='w')

        bf = tk.Frame(self.root, bg="#1a1a2e"); bf.pack(pady=10)
        bs = {"font":("Arial",12,"bold"),"bg":"#e94560","fg":"white","padx":20,"pady":8,"bd":0,"cursor":"hand2"}
        tk.Button(bf, text="Pilih Gambar", command=self.load_image, **bs).pack(side='left', padx=8)
        tk.Button(bf, text="Analisis",     command=self.run_analysis, **bs).pack(side='left', padx=8)
        tk.Button(bf, text="Scan Folder",  command=self.scan_folder, **bs).pack(side='left', padx=8)

    def load_image(self):
        p = filedialog.askopenfilename(filetypes=[("Images","*.jpg *.jpeg *.png *.bmp")])
        if not p: return
        self.current_path = p
        img   = Image.open(p).resize((350, 280))
        photo = ImageTk.PhotoImage(img)
        self.img_label.configure(image=photo, text="")
        self.img_label.image = photo

    def run_analysis(self):
        if not self.current_path:
            messagebox.showwarning("Peringatan","Pilih gambar dulu!"); return
        t0 = time.time()
        cc, cf = predict_cnn(self.current_path)
        sc     = predict_glcm_svm(self.current_path)
        elapsed = time.time() - t0
        COLOR  = {"Mutu_I":"#00cc44","Mutu_II":"#ffaa00","Tidak_Layak":"#ff3333"}
        self.cnn_result.configure(text=f"{cc}", fg=COLOR.get(cc,"#fff"))
        self.cnn_conf.configure(text=f"{cf:.1f}%")
        self.svm_result.configure(text=f"{sc}", fg=COLOR.get(sc,"#fff"))
        self.time_label.configure(text=f"{elapsed:.3f} detik")

    def scan_folder(self):
        folder = filedialog.askdirectory()
        if not folder: return
        imgs = [f for f in os.listdir(folder) if f.lower().endswith(('.jpg','.jpeg','.png','.bmp'))]
        if not imgs: messagebox.showinfo("Info","Tidak ada gambar!"); return
        rows = []
        for fname in imgs:
            try:
                cc, cf = predict_cnn(os.path.join(folder, fname))
                sc     = predict_glcm_svm(os.path.join(folder, fname))
                rows.append(f"{fname:<40} CNN: {cc:<15}({cf:.1f}%)  SVM: {sc}")
            except Exception as e:
                rows.append(f"{fname:<40} ERROR: {e}")
        out = os.path.join(folder, "hasil_sortasi.txt")
        with open(out, 'w', encoding='utf-8') as f: f.write("\n".join(rows))
        messagebox.showinfo("Selesai", f"{len(rows)} gambar diproses.\nHasil: {out}")

if __name__ == "__main__":
    root = tk.Tk()
    BawangApp(root)

  - Val Loss naik dari 0.69 ke 1.29 setelah Epoch 7

  1. Dataset kecil (~2000 train) vs CNN 3.5 juta parameter
  2. Variasi visual antar kelas sangat tinggi
  3. Arsitektur dari scratch tanpa pre-training

SOLUSI (Untuk Penelitian Lanjutan):

  A. EarlyStopping + ModelCheckpoint:
     from tensorflow.keras.callbacks import EarlyStopping, ModelCheckpoint
     early_stop = EarlyStopping(monitor='val_loss', patience=3, restore_best_weights=True)
     checkpoint = ModelCheckpoint('best_model.h5', monitor='val_accuracy', save_best_only=True)
     model.fit(..., callbacks=[early_stop, checkpoint])

  B. Transfer Learning MobileNetV2:
     from tensorflow.keras.applications import MobileNetV2
     base = MobileNetV2(weights='imagenet', include_top=False, input_shape=(224,224,3))
     base.trainable = False
     x = tf.keras.layers.GlobalAveragePooling2D()(base.output)
     x = tf.keras.layers.Dense(128, activation='relu')(x)
     out = tf.keras.layers.Dense(3, activation='softmax')(x)
     model = tf.keras.Model(inputs=base.input, outputs=out)
     # Ekspektasi akurasi: 85-95%!

  C. Augmentasi Lebih Agresif:
     datagen = ImageDataGenerator(
